module preprocess_unit(
    input clk,
    input rst,
    input [7:0] pixel_in,
    input valid_in,
    output reg [7:0] pixel_out,
    output reg valid_out
);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            pixel_out <= 0;
            valid_out <= 0;
        end else begin
            pixel_out <= pixel_in;
            valid_out <= valid_in;
        end
    end

endmodule
