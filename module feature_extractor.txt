module feature_extractor(
    input clk,
    input rst,
    input [7:0] pixel_in,
    input valid_in,
    output reg [63:0] signature,
    output reg valid_out
);

    reg [6:0] count;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            count <= 0;
            signature <= 0;
            valid_out <= 0;
        end else begin
            valid_out <= 0;

            if (valid_in) begin
                if (pixel_in > 8'd128)
                    signature[count] <= 1'b1;
                else
                    signature[count] <= 1'b0;

                count <= count + 1;
            end

            if (count == 7'd63 && valid_in) begin
                valid_out <= 1;
                count <= 0;
            end
        end
    end

endmodule
